\begin{frame}
  \frametitle{Mathematical Description}
  \begin{itemize}
    \item \textbf{Diffusion}:
      \begin{itemize}
        \item $\frac{\partial c_A}{\partial t} = D_A
          \frac{\partial^2 c_A}{\partial x^2}$
        \item $\frac{\partial c_B}{\partial t} = D_A
          \frac{\partial^2 c_B}{\partial x^2}$
      \end{itemize}

    \item \textbf{Initial Condition}: $c_A(0, x) = c_A^*, \ c_B(0, x) = c_B^*$
    \item \textbf{Infinite Boundary Conditions}:
      \begin{itemize}
        \item $c_A(t, x) \to c_A^* \text{ as } x \to \infty$
        \item $c_B(t, x) \to c_B^* \text{ as } x \to \infty$
      \end{itemize}
    \item \textbf{Electrode Boundary Condition} (Butler-Volmer):
      \begin{itemize}
        \item $D_A \frac{\partial c_A}{\partial x}\bigg|_{x=0} =
          k_0 \left[ c_A(t,0)\, e^{-\alpha \frac{F}{RT} (E(t) - E^0_f)}
          - c_B(t,0)\, e^{(1-\alpha) \frac{F}{RT} (E(t) - E^0_f)} \right]$
        \item $D_B \frac{\partial c_B}{\partial x}\bigg|_{x=0} = - D_A
          \frac{\partial c_A}{\partial x}\bigg|_{x=0}$
      \end{itemize}
    \item \textbf{Current:} $I = F A j_{0, A}$ where $ j_{0, A} = - D
      \frac{\partial c_A}{\partial x}\bigg|_{x=0}$
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Dimensionless Mathematical Description}
  Assuming $D_A = D_B$
  \begin{itemize}
    \item \textbf{Diffusion}: $\frac{\partial C_A}{\partial t} =
      \frac{\partial^2 C_A}{\partial x^2}$
    \item \textbf{Initial Condition}: $C_A(0, X) = 1$
    \item \textbf{Semi-Infinite Boundary Conditions}: $C_A(t,
      X_{\text{max}}) = 1$
    \item \textbf{Electrode Boundary Condition} (Butler-Volmer):
      $\frac{\partial C_A}{\partial X}\bigg|_{X=0} =
      K_0 \left[ C_A(T,0)\, e^{-\alpha (\theta(T) - \theta^0_f)}
      - (1 - C_A(T,0))\, e^{(1-\alpha) (\theta(T) - \theta^0_f)} \right]$
    \item \textbf{Current:} $I = \pi \epsilon F D_A  c^*_A J_{0, A}$
      where $ J_{0, A} = - D
      \frac{\partial C_A}{\partial X}\bigg|_{X=0}$
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Forward and Inverse Problem}

  \begin{itemize}
    \item Let $\phi = (\alpha, K_0, \theta_f) \in \Phi \subseteq \mathbb{R}^3$
      be the parameter vector
    \item The finite difference solver defines a map:
      $$\mathcal{P} : \Phi \to \mathcal{F}, \qquad \mathcal{P}(\phi) = J(T)$$
      where $\mathcal{F}$ is the space of dimensionless flux trajectories
    \item \textbf{Forward Problem:} Given $\phi$, compute
      $\mathcal{P}(\phi) = J(T)$
    \item \textbf{Inverse Problem:} Given noisy observations
      $y \approx J(T)$, recover
      $$\phi^* = \underset{\phi \in \Phi}{\arg\min} \;
      \|\mathcal{P}(\phi) - y\|$$
    \item The inverse problem is generally \textbf{ill-posed}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Bayesian Inference}
  \begin{itemize}
    \item \textbf{Noise model:} $y_i = \mathcal{P}(\phi)_i + \varepsilon_i,
      \quad \varepsilon_i \overset{\text{iid}}{\sim} \mathcal{N}(0, \sigma^2)$
    \item \textbf{Prior:} Uninformative uniform prior over $\Phi$,
      so $P(\phi) \propto 1$
    \item \textbf{Posterior:} $P(\phi \mid y) \propto L(y \mid \phi)
      \cdot P(\phi) \propto
      L(y \mid \phi)$
    \item \textbf{Likelihood:}
      $$L(y \mid \phi) = \prod_{i=1}^{N} \frac{1}{\sqrt{2\pi\sigma^2}}
      \exp\left(-\frac{(y_i - \mathcal{P}(\phi)_i)^2}{2\sigma^2}\right)$$
    \item \textbf{Log-likelihood:}
      \begin{align*}
        \ell(\phi \mid y) &= -\frac{N}{2}\log(2\pi\sigma^2)
        - \frac{1}{2\sigma^2}\sum_{i=1}^{N}(y_i - \mathcal{P}(\phi)_i)^2 \\
        &\propto -\frac{1}{2\sigma^2}\sum_{i=1}^{N}(y_i - \mathcal{P}(\phi)_i)^2
      \end{align*}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{MCMC}
  \begin{itemize}
    \item \textbf{Goal:} Approximate the posterior $P(\phi \mid y)$
      using samples $\{\phi^{(t)}\}_{t=1}^{T}$ from a Markov chain

    \item \textbf{Convergence:} Under mild regularity conditions,
      the empirical distribution of the chain converges to $P(\phi \mid y)$
      as $T \to \infty$

    \item \textbf{Proposal:} At each step, candidate parameters
      $\phi_{\text{cand}}$ are drawn from a proposal distribution
      $q(\phi_{\text{cand}} \mid \phi^{(t)})$

    \item \textbf{Metropolis-Hastings acceptance:} Accept
      $\phi_{\text{cand}}$ with probability
      $$r = \min\left\{ \frac{P(\phi_{\text{cand}} \mid
      y)}{P(\phi^{(t)} \mid y)}, 1 \right\}$$
      otherwise set $\phi^{(t+1)} = \phi^{(t)}$

  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Differentiable Simulator}
  \begin{itemize}
    \item \textbf{Why $\mathcal{P}$ is differentiable:} The finite
      difference discretisation reduces the PDE to a sequence of linear
      solves and pointwise nonlinear operations (Butler-Volmer), all of
      which are differentiable

    \item \textbf{Gradient of the log-likelihood:} We require
      $\nabla_\phi \ell(\phi \mid y)$

    \item \textbf{Reverse mode AD:} Since $\ell$ is scalar, reverse
      mode AD computes $\nabla_\phi \ell$ in a single backward pass
      through $\mathcal{P}$, at comparable cost to a forward evaluation

    \item \textbf{Applications:}
      \begin{itemize}
        \item \textbf{Burn-in removal:} Gradient descent on $-\ell(\phi
          \mid y)$ to initialise the chain near the posterior mode
        \item \textbf{Hamiltonian Monte Carlo:} Gradient informed MCMC
          yielding faster mixing and better exploration of $P(\phi \mid y)$
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Hamiltonian Monte Carlo}
  \begin{itemize}
    \item Augments parameter space with auxiliary momentum variables,
      using $\nabla_\phi \ell$ to make informed moves through the posterior
    \item \textbf{Effective Sample Size:} Autocorrelated samples carry
      less information than independent draws — ESS quantifies the
      equivalent number of independent samples:
      $$\text{ESS} = \frac{T}{1 + 2\sum_{k=1}^{\infty} \rho_k}$$
      where $\rho_k$ is the autocorrelation at lag $k$
  \end{itemize}
\end{frame}
